// Endpoint esperado por la actividad:
// POST /api/student/validate
// Body: { studentCode: "6B-XXXX", group: "6.º B" }
//
// IMPORTANTE:
// Ajusta los nombres de tabla/campos de la consulta SQL a tu esquema D1 real.

if (request.method === "POST" && url.pathname === "/api/student/validate") {
  try {
    const body = await request.json();
    const studentCode = String(body.studentCode || "")
      .trim()
      .toUpperCase()
      .replace(/[\u2010\u2011\u2012\u2013\u2014\u2212]/g, "-")
      .replace(/\s+/g, "");

    const group = String(body.group || "").trim();

    if (!studentCode || !group) {
      return json({ ok:false, error:"Faltan código o grupo." }, 400);
    }

    // PLANTILLA: adapta nombres de tabla/campos si tu D1 usa otros.
    const student = await env.DB.prepare(`
      SELECT
        code       AS student_code,
        name       AS student_name,
        group_name AS group_name,
        active     AS active
      FROM students
      WHERE UPPER(code) = ?
      LIMIT 1
    `).bind(studentCode).first();

    if (!student || Number(student.active ?? 1) !== 1) {
      return json({ ok:false, error:"Código no válido." }, 404);
    }

    if (String(student.group_name || "") !== group) {
      return json({ ok:false, error:"El código no corresponde al grupo seleccionado." }, 403);
    }

    return json({
      ok:true,
      studentCode: student.student_code,
      group: student.group_name,
      studentName: student.student_name || ""
    });
  } catch (err) {
    return json({ ok:false, error:"No se pudo validar el alumno." }, 500);
  }
}
